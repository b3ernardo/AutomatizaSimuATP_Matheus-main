MODEL UI2PQ3
comment--------------------------------------------------------------
This model calculates the input active and reactive power of a time 
domain voltage/current signals by a user selectable algorithm
Algorithm=0: FFT Radix2-8. More robust 
Algorithm=1: DFT recursive. Faster, some drift. Handles sub-harmonics.
One period of the signal is needed to reach steady state. 
SampleFreq frequency should be a whole multiple of the power frequency.
-----------------------------------------------------------endcomment
INPUT 
  V[1..3], I[1..3]     --input signals to be transformed to phasors
DATA  
  FREQ {DFLT:50}       --power frequency
  SampleFreq {dflt:400}--sample frequency in this model in Hz
  ScaleV {dflt:1}      --multiply output by this number 
  ScaleI {dflt:1}      --multiply output by this number
  Algorithm {dflt:0}   --0: FFT radix2-8, 1: DFT recursive
OUTPUT 
  P, Q     --Active and reactive power output signals (rms)
VAR    
  P, Q, reV[1..3],imV[1..3],reI[1..3],imI[1..3],NSAMPL, OMEGA,
  D,alpha,x1,x3,x5,x7,xre,xim,n,delta_T[0..7]

HISTORY
  V[1..3] {DFLT:0}
  I[1..3] {DFLT:0}
timestep min: 1/SampleFreq
DELAY CELLS (V[1..3]) : recip(FREQ*timestep)+2
DELAY CELLS (I[1..3]) : recip(FREQ*timestep)+2

INIT 
  OMEGA:= 2*PI*FREQ
  NSAMPL:=recip(FREQ*timestep)
  reV[1..3] :=0
  imV[1..3] :=0
  reI[1..3] :=0
  imI[1..3] :=0
  alpha:=1/sqrt(2)
  for n:= 0 to 7 do
    delta_T[n] := n/(FREQ*8)
  endfor
ENDINIT
EXEC
  P:=0
  Q:=0
  for n:=1 to 3 do
    if Algorithm=0
    then
      x1 := delay(V[n],delta_T[0],1) - delay(V[n],delta_T[4],1)
      x3 := delay(V[n],delta_T[2],1) - delay(V[n],delta_T[6],1)
      x5 := delay(V[n],delta_T[1],1) - delay(V[n],delta_T[5],1)
      x7 := delay(V[n],delta_T[3],1) - delay(V[n],delta_T[7],1)
      xRe := x1 + (x5 - x7)*alpha
      xIm := x3 + (x5 + x7)*alpha
      ReV[n] := (xRe*cos(OMEGA*T)+xIm*sin(OMEGA*T))/4
      ImV[n] := (xIm*cos(OMEGA*T)-xRe*sin(OMEGA*T))/4
      x1 := delay(I[n],delta_T[0],1) - delay(I[n],delta_T[4],1)
      x3 := delay(I[n],delta_T[2],1) - delay(I[n],delta_T[6],1)
      x5 := delay(I[n],delta_T[1],1) - delay(I[n],delta_T[5],1)
      x7 := delay(I[n],delta_T[3],1) - delay(I[n],delta_T[7],1)
      xRe := x1 + (x5 - x7)*alpha
      xIm := x3 + (x5 + x7)*alpha
      ReI[n] := (xRe*cos(OMEGA*T)+xIm*sin(OMEGA*T))/4
      ImI[n] := (xIm*cos(OMEGA*T)-xRe*sin(OMEGA*T))/4
    else
      x1:=delay(V[n],1/FREQ,2)
      D:=2/NSAMPL*(V[n]-x1)*cos(OMEGA*T) 
      ReV[n]:=ReV[n]+D
      D:=2/NSAMPL*(V[n]-x1)*sin(OMEGA*T) 
      ImV[n]:=ImV[n]-D
      x1:=delay(I[n],1/FREQ,2)
      D:=2/NSAMPL*(I[n]-x1)*cos(OMEGA*T) 
      ReI[n]:=ReI[n]+D
      D:=2/NSAMPL*(I[n]-x1)*sin(OMEGA*T) 
      ImI[n]:=ImI[n]-D
    endif
    P:=P+(reV[n]*reI[n]+imV[n]*imI[n])*ScaleV*ScaleI/2
    Q:=Q+(imV[n]*reI[n]-reV[n]*imI[n])*ScaleV*ScaleI/2
  endfor
ENDEXEC

MODEL PLLDQ
COMMENT =============================================================

    EEUG Course 2005:
    Estimation of frequency and phase by means of a PLL which
    operates using dq-components of measured voltages
    Reference: C. Zan; C. Fitzer; V. K. Ramachandaramurthy; 
    A. Arulampalam; M. Barnes; N. Jenkins: 
    Software Phased-Locked Loop applied to Dynamic Voltage 
    Restorer (DVR), Proc. IEEE-PES Winter Meeting 2001, 
    pp. 1033-1038.

    14-Aug-2005, Prof. M. Kizilcay
========================================================== ENDCOMMENT
INPUT  u[1..3]              -- phase voltages
OUTPUT FREQ
CONST
  twopi  {VAL:6.283185307}
DATA
  theta0 {DFLT:0.0}          -- phase betw. d-axis and alpha-axis
  f0     {DFLT:50.0}
  Gain   {DFLT:22.85}
VAR
  ualpha, ubeta, u0, ualfaN, ubetaN,
  u_dN, u_qN, u_d, u_q, uampl
  theta, wcon 
  omega, omega0,freq 
  ucomp, temp
HISTORY
  wcon {DFLT:0}
  u_qN {DFLT:0}
  omega {DFLT:2*PI*f0}
  INTEGRAL(omega) {DFLT:theta0*PI/180}
  freq {DFLT:50}
  theta {DFLT:theta0*PI/180}
INIT
  omega0 := 2*PI*f0
  theta := theta0*PI/180.0
ENDINIT
EXEC
  -- From three-phase to alfa-beta-zero
  ualpha := SQRT(2/3)*(u[1] - 0.5*(u[2]+u[3]))
  ubeta := SQRT(0.5)*(u[2] - u[3])
  u0 := (u[1] + u[2] + u[3])/3.0
  temp := RECIP(SQRT(ualpha*ualpha+ubeta*ubeta))
  ualfaN := ualpha*temp
  ubetaN := ubeta*temp
  -- From alfa-beta to d-q (with constant frequency)
  u_d := SQRT(2/3)*(ualpha*cos(theta) + ubeta*sin(theta))
  u_q := SQRT(2/3)*(-ualpha*sin(theta) + ubeta*cos(theta))
  uampl := SQRT(u_d*u_d+u_q*u_q)
  u_dN := ualfaN*cos(theta) + ubetaN*sin(theta)
  COMBINE ITERATE AS theta_comp
    u_qN := -ualfaN*sin(theta) + ubetaN*cos(theta)
    -- Phase-locked loop follows phase angle of the source
    CLAPLACE(wcon/u_qN) := (Gain|s0 + 0.001242|s1)/(1.0|s0 + 0.02315|s1)
    --CLAPLACE(wcon/u_qN) := (Gain|s0 + 4.0|s1)/(0.0|s0 + 1.0|s1)
    omega := omega0 + wcon
    theta := INTEGRAL(omega)
  ENDCOMBINE
  FREQ := omega/twopi
  ucomp := uampl*cos(theta)
ENDEXEC
ENDMODEL
MODEL WRITEPROBEI
comment--------------------------------------------------------------
Calculates the phasor of input V and I with FFT Radix2-8 algorithm.
The phasor is written to the LIS-file at the specified TIME.
SampleFreq should be a whole multiple of the 8* power frequency.
-----------------------------------------------------------endcomment
DATA  
  N {DFLT:3}           --number of phases
  FREQ {DFLT:50}       --power frequency
  TIME {DFLT:0}        --sample time
  SampleFreq {dflt:400}--sample freq
  IDX
INPUT 
  V[1..N], I[1..N]     --input signal to be transformed
VAR
  P[1..N], Q[1..N], Imagn[1..N],Iangl[1..N],reV[1..N],imV[1..N],reI[1..N]
  imI[1..N],OMEGA, k,alpha,x1,x3,x5,x7,xre,xim,delta_T[0..7],written,Ti,dt,DTi

HISTORY
  V[1..N] {DFLT:0}
  I[1..N] {DFLT:0}
TIMESTEP min:recip(8*SampleFreq) 
DELAY CELLS (V[1..N]) : recip(FREQ*timestep)+2
DELAY CELLS (I[1..N]) : recip(FREQ*timestep)+2

INIT 
  OMEGA:= 2*PI*FREQ
  written:=0
  reV[1..N] :=0
  imV[1..N] :=0
  reI[1..N] :=0
  imI[1..N] :=0
  alpha:=1/sqrt(2)
  for k:= 0 to 7 do
    delta_T[k] := k*recip(FREQ*8)
  endfor
  dt:=recip(SampleFreq)
  Ti:=0
ENDINIT
EXEC
  DTi:=T-Ti
  if (DTi>=0) and (written=0) then
    if (FREQ>0) and (Ti>=TIME-recip(FREQ)) then
      for k:=1 to N do
        x1 := delay(V[k],delta_T[0]+DTi,1) - delay(V[k],delta_T[4]+DTi,1)
        x3 := delay(V[k],delta_T[2]+DTi,1) - delay(V[k],delta_T[6]+DTi,1)
        x5 := delay(V[k],delta_T[1]+DTi,1) - delay(V[k],delta_T[5]+DTi,1)
        x7 := delay(V[k],delta_T[3]+DTi,1) - delay(V[k],delta_T[7]+DTi,1)
        xRe := x1 + (x5 - x7)*alpha
        xIm := x3 + (x5 + x7)*alpha
        ReV[k] := (xRe*cos(OMEGA*Ti)+xIm*sin(OMEGA*Ti))/4
        ImV[k] := (xIm*cos(OMEGA*Ti)-xRe*sin(OMEGA*Ti))/4
        x1 := delay(I[k],delta_T[0]+DTi,1) - delay(I[k],delta_T[4]+DTi,1)
        x3 := delay(I[k],delta_T[2]+DTi,1) - delay(I[k],delta_T[6]+DTi,1)
        x5 := delay(I[k],delta_T[1]+DTi,1) - delay(I[k],delta_T[5]+DTi,1)
        x7 := delay(I[k],delta_T[3]+DTi,1) - delay(I[k],delta_T[7]+DTi,1)
        xRe := x1 + (x5 - x7)*alpha
        xIm := x3 + (x5 + x7)*alpha
        ReI[k] := (xRe*cos(OMEGA*Ti)+xIm*sin(OMEGA*Ti))/4
        ImI[k] := (xIm*cos(OMEGA*Ti)-xRe*sin(OMEGA*Ti))/4
      endfor
    endif 
    if Ti>=TIME then
      if FREQ>0 then
        for k:=1 to N do
          P[k]:=(reV[k]*reI[k]+imV[k]*imI[k])/2
          Q[k]:=(imV[k]*reI[k]-reV[k]*imI[k])/2
          Imagn[k]:=sqrt(reI[k]**2+imI[k]**2)
          if (imI[k]=0) and (reI[k]=0) then
            Iangl[k]:=0
          else
            Iangl[k]:=atan2(imI[k],reI[k])*180/PI
          endif
        endfor 
      else
        for k:=1 to N do
          P[k]:=I[k]*V[k]/2
          Q[k]:=0
          Imagn[k]:=I[k]
          Iangl[k]:=0
        endfor
      endif
      write('BEGIN WRITE @WRITEPROBEI',trunc(Idx))    --Do not modify this line, replaced by ATPDraw
      for k:=1 to N do
        write(Imagn[k],' ', Iangl[k],' ', P[k],' ' ,Q[k],' ' , TIME)            
            --Add multiple columns or multiple lines 
      endfor
      write('END WRITE @WRITEPROBEI',trunc(Idx))      --Do not modify this line, replaced by ATPDraw
      written:=1
    endif
    Ti:=Ti+dt
  endif
ENDEXEC
ENDMODEL
MODEL WRITEPROBEV
comment---------------------------------------------------------
Calculates the phasor of input V with FFT Radix2-8 algorithm.
The phasor is written to the LIS-file at the specified TIME.
SampleFreq should be a whole multiple of the 8* power frequency.
-------------------------------------------------------endcomment
DATA  
  N {DFLT:3}           --number of phases
  FREQ {DFLT:50}       --power frequency
  TIME {DFLT:0}        --sample time
  SampleFreq {dflt:400}--sample freq
  IDX
INPUT 
  V[1..N]     --input signal to be transformed
VAR    
  Vmagn[1..N],Vangl[1..N],reV[1..N],imV[1..N],OMEGA,Ti,dt,DTi
  k,alpha,x1,x3,x5,x7,xre,xim,delta_T[0..7],written

HISTORY
  V[1..N] {DFLT:0}
TIMESTEP min: recip(8*SampleFreq) 
DELAY CELLS (V[1..N]) : recip(FREQ*timestep)+2

INIT 
  OMEGA:= 2*PI*FREQ
  written:=0
  reV[1..N] :=0
  imV[1..N] :=0
  alpha:=1/sqrt(2)
  for k:= 0 to 7 do
    delta_T[k] := k*recip(FREQ*8)
  endfor
  dt:=recip(SampleFreq)
  Ti:=0
ENDINIT
EXEC
  DTi:=T-Ti
  if (DTi>=0) and (written=0) then
    if (FREQ>0) and (Ti>=TIME-recip(FREQ)) then
      for k:=1 to N do
        x1 := delay(V[k],delta_T[0]+DTi,1) - delay(V[k],delta_T[4]+DTi,1)
        x3 := delay(V[k],delta_T[2]+DTi,1) - delay(V[k],delta_T[6]+DTi,1)
        x5 := delay(V[k],delta_T[1]+DTi,1) - delay(V[k],delta_T[5]+DTi,1)
        x7 := delay(V[k],delta_T[3]+DTi,1) - delay(V[k],delta_T[7]+DTi,1)
        xRe := x1 + (x5 - x7)*alpha
        xIm := x3 + (x5 + x7)*alpha
        ReV[k] := (xRe*cos(OMEGA*Ti)+xIm*sin(OMEGA*Ti))/4
        ImV[k] := (xIm*cos(OMEGA*Ti)-xRe*sin(OMEGA*Ti))/4
      endfor
    endif 
    if Ti>=TIME then
      if FREQ>0 then
        for k:=1 to N do
          Vmagn[k]:=sqrt(reV[k]**2+imV[k]**2)
          if (imV[k]=0) and (reV[k]=0) then
            Vangl[k]:=0
          else
            Vangl[k]:=atan2(imV[k],reV[k])*180/PI
          endif
        endfor 
      else
        for k:=1 to N do
          Vmagn[k]:=V[k]
          Vangl[k]:=0
        endfor
      endif
      write('BEGIN WRITE @WRITEPROBEV',trunc(Idx))    --Do not modify this line, replaced by ATPDraw
      for k:=1 to N do
        write(Vmagn[k],' ', Vangl[k],' ', TIME)       --Add multiple columns or multiple lines 
      endfor
      write('END WRITE @WRITEPROBEV',trunc(Idx))      --Do not modify this line, replaced by ATPDraw
      written:=1
    endif
    Ti:=Ti+dt
  endif
ENDEXEC
ENDMODEL
MODEL PVCONTROLMED
COMMENT =============================================================
    09-May-2023, Thiago Souza Menezes
========================================================== ENDCOMMENT
INPUT  volt[1..3] -- phase voltages
       curr[1..3] -- line current out of inverter            
OUTPUT Vref[1..3]
CONST
  twopi  {VAL:6.283185307}, a  {VAL:2.094395102}
DATA
  S      {dflt:1}     --MVA
  Vrms   {dflt:1}     --kV L-L rms
  Pref   {dflt:1}     --pu
  Qref   {dflt:0.1}   --pu
  Freq0  {dflt:60}    --Hz
  swfreq {dflt:3840}  --Hz
  Kpe    {dflt:0.15}  --Kp ext
  Kie    {dflt:10}    --Ki ext 
  Kpi    {dflt:40.512}--Kp int
  Kii    {dflt:42.24}--Ki int
  Lf     {dflt:21.1}  --mH filter inductance
  theta0 {DFLT:0.0}  --initial phase betw. d-axis and alpha-axis
  TPV    {dflt:0.0}     --tempo de entrada do módulo PV
VAR
  UD, UQ, ID, IQ, UDF, UQF, IDF, IQF, IDerr, IQerr, VDCor, VQcor, tx, triang 
  VDref, VQref, Vref[1..3], freq, omega, theta, i
  P, Q, IDref, IQref, Vlim, Lfpu, Perr, Qerr
  Ualfa, Ubeta, Upeak, Uquad, Dfrq, KPLL, TPLL, TF2A, modulo, angle, Vdnew
  Vqnew, modcur, angcur, Ilim
TIMESTEP min:recip(swfreq)
HISTORY
  omega {DFLT:theta0*PI/180}
  theta {dflt:0.0}
  freq {DFLT:Freq0}
  UDF {dflt:1}
  UQF {dflt:0}
  IDF {dflt:1}
  IQF {dflt:0}
  VDcor {dflt:1}
  VQcor {dflt:0}
  Perr {dflt:0}
  Qerr {dflt:0}
  IDref {dflt:0}
  IQref {dflt:0}
  Ualfa {dflt: 0.0}
  Ubeta {dflt: 0.0}
  Uquad {dflt: 0.0}
  Dfrq {dflt: 0.0}
  P {dflt:0}
  Q {dflt:0}
INIT
  omega:=twopi*Freq0
  theta := theta0*PI/180.0
  Lfpu:=Lf/1000        --scale to H
  Vlim:=1.1
  Ilim:= 0.376
  KPLL := 10.0
  TPLL := 0.00125
  TF2A := 0.0000070361933--1.0/(2.0*twopi*Freq0)**2 -- 2a harm filter
  modulo:=0
  angle:=0
  modcur:=0
  angcur:=0
  IDref:=0
  VDref:=1
  VQref:=0
  Perr:= 0
  Qerr:= 0
  tx:=0
  triang:=0
  Vref[1..3]:=0
ENDINIT
EXEC
  if t >= TPV then  
      -----------------------------------------------------------------------------
      -- Park transforms
      UD:=volt[1]*cos(theta)+volt[2]*cos(theta-a)+volt[3]*cos(theta+a) --Park
      UQ:=-volt[1]*sin(theta)-volt[2]*sin(theta-a)-volt[3]*sin(theta+a)--Park
      ID:=curr[1]*cos(theta)+curr[2]*cos(theta-a)+curr[3]*cos(theta+a) --Park
      IQ:=-curr[1]*sin(theta)-curr[2]*sin(theta-a)-curr[3]*sin(theta+a)--Park
      claplace(UDF/UD):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      claplace(UQF/UQ):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz  
      claplace(IDF/ID):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      claplace(IQF/IQ):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      -- calculo das potencias do inversor
      P:= (UDF*IDF+UQF*IQF)*1.5
      Q:= -(UDF*IQF-UQF*IDF)*1.5
      -- filtrar para reduzir as oscilações na potencia
      claplace(P/P):=(1|s0)/(1|s0+0.006366|s1) --LP filter 25hz
      claplace(Q/Q):=(1|s0)/(1|s0+0.006366|s1) --LP filter 25hz
      -----------------------------------------------------------------------------  
      -- malha externa de potencia
      Perr := Pref - P/(S*1E6)    -- em pu
      claplace(IDref/Perr) := (Kpe*Kie|s0 + Kpe|s1)/(0.0|s0 + 1.0|s1)--PI
      Qerr := Qref - Q/(S*1E6)    -- em pu
      claplace(IQref/Qerr) := (-Kpe*Kie|s0 - Kpe|s1)/(0.0|s0 + 1.0|s1)--PI  
      -- saturacao do modulo de id e iq para injetar o maximo de 2 pu
      modcur := sqrt(IDref*IDref+IQref*IQref)
      if (abs(IDref)>1E-6) OR (abs(IQref)>1E-6) then
        angcur := atan2(IQref,IDref)
      else
        angcur:=0
      endif
      if modcur >= Ilim then
        modcur:=Ilim
      endif
      if modcur < 0 then
        modcur:=0
      endif
      IQref := modcur*sin(angcur)
      IDref := modcur*cos(angcur)
      -----------------------------------------------------------------------------
      -- malha interna de corrente
      IDerr:=(IDref-IDF/1000.0)
      IQerr:=(IQref-IQF/1000.0)
      claplace(VDcor/IDerr) :=(Kpi*Kii|S0 + Kpi|S1)/(0.0|S0 + 1.0|S1) --PI
      claplace(VQcor/IQerr) :=(Kpi*Kii|S0 + Kpi|S1)/(0.0|S0 + 1.0|S1) --PI
      VDref:=(UDF/1000.0-(IQF/1000.0)*omega*Lfpu+VDcor)*(0.222222222) --2000/9000
      VQref:=(UQF/1000.0+(IDF/1000.0)*omega*Lfpu+VQcor)*(0.222222222) --2000/9000
      -- saturacao do modulo de vd e vq
      modulo := sqrt(VDref*VDref+VQref*VQref)
      if (abs(VDref)>1E-6) OR (abs(VQref)>1E-6) then
        angle := atan2(VQref,VDref)
      else
        angle:=0
      endif
      if modulo >= Vlim then
        modulo:=Vlim
      endif
      if modulo < 0 then
        modulo:=0
      endif
      Vqnew := 4500*modulo*sin(angle)
      Vdnew := 4500*modulo*cos(angle)
      -----------------------------------------------------------------------------
      -- gerar sinais de tensão ABC para as fontes  
      Vref[1]:=(Vdnew*cos(theta)-Vqnew*sin(theta))
      Vref[2]:=(Vdnew*cos(theta-a)-Vqnew*sin(theta-a))
      Vref[3]:=(Vdnew*cos(theta+a)-Vqnew*sin(theta+a))
      -----------------------------------------------------------------------------
      -- PLL usando alfa beta
      -- clarke
      Ualfa := 0.333333333*(2.0*volt[1] - volt[2] - volt[3]) -- /3
      Ubeta := 0.577350269*(volt[2] - volt[3]) -- sqrt(3)/3
      claplace(Ualfa/Ualfa):=((1.0)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      claplace(Ubeta/Ubeta):=((1.0)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz 
      -- normalizar os valores
      Upeak := sqrt(Ualfa*Ualfa + Ubeta*Ubeta)
      Ualfa := Ualfa / Upeak
      Ubeta := Ubeta / Upeak
      -- converter para dq
      Uquad := KPLL*(Ubeta * cos(theta) - Ualfa * sin(theta))
      claplace(Dfrq/Uquad) := (1.0|s0 + TF2A|s2)/(1.0|s0 + TPLL|s1 + TF2A|s2)
      -- calcular a frequencia instantanea no sinal
      freq := Freq0 + Dfrq
      claplace(freq/freq):=(1|s0)/(1|s0+0.005305164|s1) --LP filter 30hz
      omega := twopi * freq
      -- integrar a velocida angular para calcular o angulo de fase
      claplace(theta/omega) := (1.0|s0)/(1.0|s1)--integral de wt
      -- limitar theta entre 0 e dois PI  
      if theta>twopi then
        theta:=theta-twopi
      endif
    endif
ENDEXEC
ENDMODEL
MODEL BESSCONTROL

    09-May-2023, Thiago Souza Menezes

       curr[1..3] -- line current out of inverter            
OUTPUT Vref[1..3]
CONST
  twopi  {VAL:6.283185307}, a  {VAL:2.094395102}
DATA
  S      {dflt:1}     --MVA
  Vrms   {dflt:1}     --kV L-L rms
  Vset   {dflt:1}     --pu
  Fset   {dflt:1}     --pu
  Freq0  {dflt:60}    --Hz
  swfreq {dflt:3840}  --Hz
  Kpe    {dflt:0.15}  --Kp ext
  Kie    {dflt:10}    --Ki ext 
  Kpi    {dflt:40}    --Kp int
  Kii    {dflt:45}    --Ki int
  Lf     {dflt:21.1}  --mH filter inductance
  theta0 {DFLT:0.0}   --initial phase betw. d-axis and alpha-axis
  TPV    {dflt:0.0}     --tempo de entrada do módulo PV
VAR
  UD, UQ, ID, IQ, UDF, UQF, IDF, IQF, IDerr, IQerr, VDCor, VQcor, tx, triang 
  VDref, VQref, Vref[1..3], freq, freqf, omega, theta, i, Vpu, Vmed, Vmedf
  P, Q, Sva, Pref, Qref, Pref2, Qref2, IDref, IQref, Vlim, Lfpu, Perr, Qerr
  Ualfa, Ubeta, Upeak, Uquad, Dfrq, KPLL, TPLL, TF2A, modulo, angle, Vdnew, 
Vqnew
  modcur, angcur, Ilim, dt, Period
TIMESTEP min:recip(swfreq)
DELAY CELLS (freq,freqf): (swfreq/60)+2
DELAY CELLS (Vmed,Vmedf): (swfreq/60)+2
HISTORY
  omega {DFLT:theta0*PI/180}
  theta {dflt:0.0}
  freq {DFLT:Freq0}
  freqf {DFLT:Freq0}
  UDF {dflt:1}
  UQF {dflt:0}
  IDF {dflt:1}
  IQF {dflt:0}
  VDcor {dflt:1}
  VQcor {dflt:0}
  Vmed {dflt:1}
  Vmedf {dflt:1}
  Perr {dflt:0}
  Qerr {dflt:0}
  IDref {dflt:0}
  IQref {dflt:0}
  Ualfa {dflt: 0.0}
  Ubeta {dflt: 0.0}
  Uquad {dflt: 0.0}
  Dfrq {dflt: 0.0}
  P {dflt:0}
  Q {dflt:0}
  Sva {dflt:0}
  Pref2 {dflt:0}
  Qref2 {dflt:0}
INIT
  omega:=twopi*Freq0
  theta := theta0*PI/180.0
  Lfpu:=Lf/1000        --scale to H
  Vlim:=1.1
  Ilim:= 2*0.19341
  KPLL := 10.0
  TPLL := 0.00125
  TF2A := 0.0000070361933--1.0/(2.0*twopi*Freq0)**2 -- 2a harm filter
  modulo:=0
  angle:=0
  modcur:=0
  angcur:=0
  IDref:=0
  VDref:=1
  VQref:=0
  Perr:= 0
  Qerr:= 0
  tx:=0
  triang:=0
  Vref[1..3]:=0
  Vpu:=Vrms*1000*sqrt(2/3)
  Vmed:=Vpu
  Vmedf:=Vpu
  Pref2:=0.1
  Qref2:=0.1
  freq:=Freq0
  dt:=recip(swfreq)
  Period:=recip(Freq0)
ENDINIT
EXEC
  if t >= TPV then
      -----------------------------------------------------------------------------
      --Park transforms
      UD:=volt[1]*cos(theta)+volt[2]*cos(theta-a)+volt[3]*cos(theta+a) --Park
      UQ:=-volt[1]*sin(theta)-volt[2]*sin(theta-a)-volt[3]*sin(theta+a)--Park
      ID:=curr[1]*cos(theta)+curr[2]*cos(theta-a)+curr[3]*cos(theta+a) --Park
      IQ:=-curr[1]*sin(theta)-curr[2]*sin(theta-a)-curr[3]*sin(theta+a)--Park
      claplace(UDF/UD):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      claplace(UQF/UQ):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz  
      claplace(IDF/ID):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      claplace(IQF/IQ):=((0.666666667)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz 
      -- calculo das potencias do inversor
      P:= (UDF*IDF+UQF*IQF)*1.5
      Q:= -(UDF*IQF-UQF*IDF)*1.5  
      claplace(P/P):=(1|s0)/(1|s0+0.006366|s1)--LP filter 25hz
      claplace(Q/Q):=(1|s0)/(1|s0+0.006366|s1)--LP filter 25hz  
      Sva:=sqrt(P*P+Q*Q)
      -----------------------------------------------------------------------------
      -- Calculo das malhas do controle droop
      -- Esperar o sistema estabilizar antes de iniciar o droop da bateria
      if t >= TPV + 0.2 then
        -- potencia ativa de ref 
        -- filtra a frequencia
        freqf := freqf + (delay(freq, dt, 2) - delay(freq, Period+dt, 2))*
recip(swfreq/60)
        Pref := (Fset - (freqf/60))
        Pref2:=Pref*2     -- apenas ganho proporcional do droop (alterado de 6 p/ 2)
        ---------------------------------------------------------------------------
        -- potencia reativa de ref 
        Vmed:=0.666666667*UD
        -- filtra a tensao
        Vmedf := Vmedf + (delay(Vmed, dt, 2) - delay(Vmed, Period+dt, 2))*
recip(swfreq/60)
        Qref := (Vset - (Vmedf/Vpu))
        Qref2:=Qref*2    -- apenas ganho proporcional do droop (alterado de 6 p/ 2)
      else -- inicializa com injeção zero
        Pref2:=0
        Qref2:=0
      endif
      -----------------------------------------------------------------------------
      -- malha externa de potencia
      Perr := Pref2 - P/(S*1E6)    -- em pu
      claplace(IDref/Perr) := (Kpe*Kie|s0 + Kpe|s1)/(0.0|s0 + 1.0|s1) --PI regulator
      Qerr := Qref2 - Q/(S*1E6)    -- em pu
      claplace(IQref/Qerr) := (-Kpe*Kie|s0 - Kpe|s1)/(0.0|s0 + 1.0|s1) --PI regulator    
      -- saturacao do modulo de id e iq para injetar o maximo de 2 pu
      modcur := sqrt(IDref*IDref+IQref*IQref)  
      if (abs(IDref)>1E-6) OR (abs(IQref)>1E-6) then
        angcur := atan2(IQref,IDref)
      else
        angcur:=0
      endif
      if modcur >= Ilim then
        modcur:=Ilim
      endif
      if modcur < 0 then
        modcur:=0
      endif
      IQref := modcur*sin(angcur)
      IDref := modcur*cos(angcur)  
      -----------------------------------------------------------------------------
      -- malha interna de corrente
      IDerr:=(IDref-IDF/1000.0)
      IQerr:=(IQref-IQF/1000.0)
      claplace(VDcor/IDerr) :=(Kpi*Kii|S0 + Kpi|S1)/(0.0|S0 + 1.0|S1) --PI regulator
      claplace(VQcor/IQerr) :=(Kpi*Kii|S0 + Kpi|S1)/(0.0|S0 + 1.0|S1) --PI regulator  
      VDref:=(UDF/1000.0-(IQF/1000.0)*omega*Lfpu+VDcor)*(0.222222222) --2000/9000
      VQref:=(UQF/1000.0+(IDF/1000.0)*omega*Lfpu+VQcor)*(0.222222222) --2000/9000  
      ----------------------------------------------    
      -- saturacao do modulo de vd e vq
      modulo := sqrt(VDref*VDref+VQref*VQref)  
      if (abs(VDref)>1E-6) OR (abs(VQref)>1E-6) then
        angle := atan2(VQref,VDref)
      else
        angle:=0
      endif
      if modulo >= Vlim then
        modulo:=Vlim
      endif
      if modulo < 0 then
        modulo:=0
      endif
      Vqnew := 4500*modulo*sin(angle)
      Vdnew := 4500*modulo*cos(angle)
      -----------------------------------------------------------------------------
      -- gerar os sinais de tensao ABC para as fontes  
      Vref[1]:=(Vdnew*cos(theta)-Vqnew*sin(theta))
      Vref[2]:=(Vdnew*cos(theta-a)-Vqnew*sin(theta-a))
      Vref[3]:=(Vdnew*cos(theta+a)-Vqnew*sin(theta+a))
      -----------------------------------------------------------------------------
      -- PLL usando alfa beta
      -- clarke
      Ualfa := 0.333333333*(2.0*volt[1] - volt[2] - volt[3]) -- /3
      Ubeta := 0.577350269*(volt[2] - volt[3]) -- sqrt(3)/3
      claplace(Ualfa/Ualfa):=((1.0)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz
      claplace(Ubeta/Ubeta):=((1.0)|s0)/(1|s0+0.00016|s1)  --LP filter 1kHz 
      -- normalizar os valores
      Upeak := sqrt(Ualfa*Ualfa + Ubeta*Ubeta)
      Ualfa := Ualfa / Upeak
      Ubeta := Ubeta / Upeak
      -- converter para dq
      Uquad := KPLL*(Ubeta * cos(theta) - Ualfa * sin(theta))
      claplace(Dfrq/Uquad) := (1.0|s0 + TF2A|s2)/(1.0|s0 + TPLL|s1 + TF2A|s2)
      -- calcular a frequencia instantanea no sinal
      freq := Freq0 + Dfrq
      claplace(freq/freq):=(1|s0)/(1|s0+0.005305164|s1) --LP filter 30hz
      omega := twopi * freq
      -- integrar a velocida angular para calcular o angulo de fase
      claplace(theta/omega) := (1.0|s0)/(1.0|s1)--integral de wt
      -- limitar theta entre 0 e dois PI    
      if theta>twopi then
        theta:=theta-twopi
      endif  
    endif
ENDEXEC
ENDMODEL

    r3ModPos:= ((vfalta/sqrt(3))/(ifalta * (1+ModPos/100)))/2   --r3Ss - ((vfalta/sqrt(3))/(ifalta * (1+ModPos/100)))/2   
    r3ModNeg:= ((vfalta/sqrt(3))/(ifalta * (1-ModNeg/100)))/2 -- - r3Ss
    cont_cic:=0
    num_cic:=0
    
    
    -- Inicialização do Modelo R2
    R2:=400
    r2Ss:=((vfalta/sqrt(3))/ifalta)/2
    r2Spike:= r2Ss - ((vfalta/sqrt(3))/(ifalta * (1+iSpike/100)))/2
    
    --Inicialização do Modelo R1
    R1:=0
    tempo_exponencial:=0
    IF iperc=0 THEN
        iperctemp:=10
    ELSE
        iperctemp:=iperc
    ENDIF
    L := 1/(ln(cbup/freq + 1))
    A := 100*(r2Ss+r3SS)/iperctemp - (r2Ss+r3Ss)
    
    -- Inicialização da Resistência de falta e do Modelo de Cassie "Normalizado"
    Rf:=10000

      IF prevval(tensao)<0 AND tensao>=0 THEN ciclo:=true semiciclo:=true ENDIF
  ENDIF --aguarda 2 ciclos (transitórios)
  
  
  -- Verifica a ruptura do condutor
  IF t >= ruptt THEN
     rompe:=true
  ENDIF
  -- Verifica o toque do condutor 
  IF t > toucht AND ciclo THEN -- Espera um novo ciclo para iniciar o Build-up
     toca:=true
  ENDIF

  -- Abre a fase afetada
  IF rompe=true THEN
      CH1:=0
  ENDIF

  -- Toque do cabo e intermitência da falta
  IF toca=true AND prevval(toca)=false THEN
      CH2:=1
      req:=true
  ENDIF

      IF ciclo=true THEN  -- Conta os ciclos 
          cont_ciclo:=cont_ciclo+1
      ENDIF
      IF CH2=1 AND cont_ciclo=c_on THEN -- Ciclos em que existe arco  
          CH2:=0
          cont_ciclo:=0
      ENDIF
      IF CH2=0 AND cont_ciclo=c_off THEN -- Ciclos em que o arco foi extinto
          CH2:=1
          cont_ciclo:=0
          req:=true

          -- Reiniciando as resistências para aleatorizar o evento na intermitência
          rand := (1.2-0.1)*random() + 0.1
          r2Ss := ((vfalta/sqrt(3))/(rand*ifalta))/2
          r3Ss := ((vfalta/sqrt(3))/(rand*ifalta))/2
          
          -- Iniciando um novo Build-up
          IF iperc=0 THEN
              iperctemp:=10
          ELSE
              iperctemp:=random()*iperc

          A := 100*(r2Ss+r3Ss)/iperctemp - (r2Ss+r3Ss) 
      ENDIF

          R1:=0
      ELSE
          IF ciclo=true THEN

              R1 := A * (1/ln(tempo_exponencial + 1) - L)/(1/ln(1/60 + 1) - L)
          ENDIF
      ENDIF

      IF semiciclo THEN -- A assimetria e o Spike é para novos semiciclos

          IF rand>=0 AND rand<(1-(Ipasy+IpSpike)/100) THEN

      ENDIF

          IF rand>=0 AND rand<1-(pModPos+pModNeg)/100 THEN

          ELSE

              R3_lim := randomizar*(random()*(r3ModNeg))+(1-randomizar)*r3ModNeg
          ENDIF
          R3_decai_random:= (1.2-0.1)*random() + 0.1
      ENDIF
      
      -- Faz a variação da modulação ser suave
      IF R3 <> R3_lim THEN
          R3 := R3 + (R3_lim - R3)/(R3_decai_random*decaimento) -- A suavidade pode ser ajustada
      ENDIF

      -- Calcula a resistência equivalente de cada característica
      Rref := R1 + R2 + R3
      
      -- Modelo Cassie "Normalizado"
      g:=exp( rand_assimetria*(1/(vfalta/sqrt(3))) * (abs(tensao) + 
abs(prevval(tensao)))/2 - 1 )
      
      -- Modula a resistência de falta de acordo com as características do evento e do modelo de Cassie
      Rf:=Rref*(1/g)
  ELSE
      Rf:=1000000
      tempo_exponencial:=0
  ENDIF

ENDEXEC
ENDMODEL
